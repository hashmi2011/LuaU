------------------------------------------------
-- CHARACTER HELPERS
-- Small helper functions for finding and updating the player.
------------------------------------------------
local function getHumanoid()
    local char = player.Character
    if not char then return nil end

    return char:FindFirstChildOfClass("Humanoid")
end

local function applySpeed()
    local hum = getHumanoid()
    if not hum then return end

    -- When custom speed is off, leave WalkSpeed entirely under the game's control.
    if enabled and hum.WalkSpeed ~= speed then
        local targetSpeed = speed
        hum.WalkSpeed = targetSpeed
    end
end

------------------------------------------------
-- SPEED WATCHDOG
-- Keeps speed applied if another local script changes WalkSpeed.
------------------------------------------------
local function watchHumanoid(hum)
    if walkSpeedConn then
        walkSpeedConn:Disconnect()
        walkSpeedConn = nil
    end

    if not hum then return end

    walkSpeedConn = hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
        if enabled then
            task.defer(applySpeed)
        else
            -- Keep the slider's minimum aligned with the speed selected by the game.
            baseSpeed = hum.WalkSpeed
            if not hasCustomSpeed then
                speed = baseSpeed
            end
            task.defer(function()
                setMaxSpeedValue(baseSpeed + MAX_SPEED_BONUS)
                updateSpeedSlider()
            end)
        end
    end)

    if not enabled then
        baseSpeed = hum.WalkSpeed
        if not hasCustomSpeed then
            speed = baseSpeed
        end
    end

    applySpeed()
end

local function startSpeedWatchdog()
    if speedConn then
        speedConn:Disconnect()
    end

    speedConn = RunService.Heartbeat:Connect(function()
        applySpeed()
    end)

    if charConn then
        charConn:Disconnect()
    end

    charConn = player.CharacterAdded:Connect(function(char)
        local hum = char:WaitForChild("Humanoid", 5)
        watchHumanoid(hum)
    end)

    if player.Character then
        watchHumanoid(player.Character:FindFirstChildOfClass("Humanoid"))
    end
end

------------------------------------------------
-- SLIDER LOGIC
-- Updates the speed when the slider is clicked or dragged.
------------------------------------------------
function updateSpeedSlider()
    local rel = 0

    if MAX_SPEED_BONUS > 0 then
        rel = math.clamp((speed - baseSpeed) / MAX_SPEED_BONUS, 0, 1)
    end

    fill.Size = UDim2.new(rel, 0, 1, 0)
    label.Text = "Speed: " .. speed
end

local function updateFlySpeedSlider()
    local maxSpeed = baseSpeed + MAX_SPEED_BONUS
    local rel = 0

    if maxSpeed > FLY_MIN_SPEED then
        rel = math.clamp((flySpeed - FLY_MIN_SPEED) / (maxSpeed - FLY_MIN_SPEED), 0, 1)
    end

    flyFill.Size = UDim2.new(rel, 0, 1, 0)
    flySpeedLabel.Text = "Fly Speed: " .. flySpeed
end

local function setSpeed(x)
    local rel = math.clamp(
        (x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X,
        0,
        1
    )

    speed = math.floor(baseSpeed + rel * MAX_SPEED_BONUS)
    hasCustomSpeed = true
    updateSpeedSlider()

    applySpeed()
end

local function setFlySpeed(x)
    local maxSpeed = baseSpeed + MAX_SPEED_BONUS
    local rel = math.clamp(
        (x - flyBar.AbsolutePosition.X) / flyBar.AbsoluteSize.X,
        0,
        1
    )

    flySpeed = math.floor(FLY_MIN_SPEED + rel * (maxSpeed - FLY_MIN_SPEED))
    updateFlySpeedSlider()
end

local function setMaxSpeed(x)
    local rel = math.clamp(
        (x - maxBar.AbsolutePosition.X) / maxBar.AbsoluteSize.X,
        0,
        1
    )

    local maxSpeed = math.floor(baseSpeed + rel * (MAX_ALLOWED_SPEED - baseSpeed))
    MAX_SPEED_BONUS = maxSpeed - baseSpeed
    maxFill.Size = UDim2.new(rel, 0, 1, 0)
    maxSpeedLabel.Text = "Max Speed: " .. maxSpeed
    maxSpeedBox.Text = tostring(maxSpeed)

    if speed > maxSpeed then
        speed = maxSpeed
    end

    if flySpeed > maxSpeed then
        flySpeed = maxSpeed
    end

    updateSpeedSlider()
    updateFlySpeedSlider()
    applySpeed()
end

function setMaxSpeedValue(value)
    local maxSpeed = math.floor(value)
    local rel = math.clamp(
        (maxSpeed - baseSpeed) / (MAX_ALLOWED_SPEED - baseSpeed),
        0,
        1
    )

    MAX_SPEED_BONUS = maxSpeed - baseSpeed
    maxFill.Size = UDim2.new(rel, 0, 1, 0)
    maxSpeedLabel.Text = "Max Speed: " .. maxSpeed
    maxSpeedBox.Text = tostring(maxSpeed)

    if speed > maxSpeed then
        speed = maxSpeed
    end

    if flySpeed > maxSpeed then
        flySpeed = maxSpeed
    end

    updateSpeedSlider()
    updateFlySpeedSlider()
    applySpeed()
end

------------------------------------------------
-- FLY FUNCTIONS
-- Creates, protects, or removes the BodyVelocity and BodyGyro used for fly.
------------------------------------------------
local function ensureFlyParts()
    if not flying then return false end

    local char = player.Character
    if not char then return false end

    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")

    if not hrp or not hum then return false end

    if not hum.PlatformStand then
        hum.PlatformStand = true
    end

    if not vel or vel.Parent ~= hrp then
        if vel then
            vel:Destroy()
        end

        vel = Instance.new("BodyVelocity")
        vel.Name = "ProtectedFlyVelocity"
        vel.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        vel.Parent = hrp
    end

    if not gyro or gyro.Parent ~= hrp then
        if gyro then
            gyro:Destroy()
        end

        gyro = Instance.new("BodyGyro")
        gyro.Name = "ProtectedFlyGyro"
        gyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        gyro.Parent = hrp
    end

    return true
end

local function stopFly()
    flying = false

    if flyConn then
        flyConn:Disconnect()
        flyConn = nil
    end

    if flyWatchConn then
        flyWatchConn:Disconnect()
        flyWatchConn = nil
    end

    if vel then
        vel:Destroy()
        vel = nil
    end

    if gyro then
        gyro:Destroy()
        gyro = nil
    end

    local hum = getHumanoid()
    if hum then
        hum.PlatformStand = false
    end
end

local function startFly()
    flying = true

    if not ensureFlyParts() then
        flying = false
        return false
    end

    if flyWatchConn then
        flyWatchConn:Disconnect()
    end

    flyWatchConn = RunService.Heartbeat:Connect(function()
        ensureFlyParts()
    end)

    if flyConn then
        flyConn:Disconnect()
    end

    flyConn = RunService.RenderStepped:Connect(function()
        if not flying then return end
        if not ensureFlyParts() then return end

        local cam = workspace.CurrentCamera
        local dir = Vector3.zero

        if UIS:IsKeyDown(Enum.KeyCode.W) then
            dir += cam.CFrame.LookVector
        end

        if UIS:IsKeyDown(Enum.KeyCode.S) then
            dir -= cam.CFrame.LookVector
        end

        if UIS:IsKeyDown(Enum.KeyCode.A) then
            dir -= cam.CFrame.RightVector
        end

        if UIS:IsKeyDown(Enum.KeyCode.D) then
            dir += cam.CFrame.RightVector
        end

        if UIS:IsKeyDown(Enum.KeyCode.Space) then
            dir += Vector3.new(0, 1, 0)
        end

        if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
            dir -= Vector3.new(0, 1, 0)
        end

        vel.Velocity =
            dir.Magnitude > 0
            and dir.Unit * flySpeed
            or Vector3.zero

        gyro.CFrame = cam.CFrame
    end)

    return true
end

------------------------------------------------
-- ESP FUNCTIONS
-- Adds highlights to other players so they are visible through walls.
------------------------------------------------
local function removeESPFromPlayer(targetPlayer)
    local highlight = espHighlights[targetPlayer]
    if highlight then
        highlight:Destroy()
        espHighlights[targetPlayer] = nil
    end
end

local function isPlayerInESPRange(targetPlayer)
    -- A range of 0 means unlimited distance.
    if espRange <= 0 then return true end

    local ownChar = player.Character
    local targetChar = targetPlayer.Character
    local ownRoot = ownChar and ownChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")

    if not ownRoot or not targetRoot then return false end

    return (ownRoot.Position - targetRoot.Position).Magnitude <= espRange
end

local function addESPToPlayer(targetPlayer)
    if targetPlayer == player then return end
    if not espEnabled then return end
    if teamCheckEnabled and player.Team ~= nil and targetPlayer.Team == player.Team then return end

    if not isPlayerInESPRange(targetPlayer) then
        removeESPFromPlayer(targetPlayer)
        return
    end

    local char = targetPlayer.Character
    if not char then return end

    local existing = espHighlights[targetPlayer]
    if existing and existing.Parent == char then
        existing.FillColor = espColor
        return
    end

    removeESPFromPlayer(targetPlayer)

    local highlight = Instance.new("Highlight")
    highlight.Name = "PlayerESP"
    highlight.Adornee = char
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.FillColor = espColor
    highlight.FillTransparency = 0.75
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.OutlineTransparency = 0
    highlight.Parent = char

    espHighlights[targetPlayer] = highlight
end

local function disconnectESPConnections()
    for _, conn in ipairs(espConnections) do
        conn:Disconnect()
    end

    table.clear(espConnections)
end

local function stopESP()
    espEnabled = false
    espBtn.Text = "ESP: OFF"

    disconnectESPConnections()

    for targetPlayer in pairs(espHighlights) do
        removeESPFromPlayer(targetPlayer)
    end
end

local function startESP()
    espEnabled = true
    espBtn.Text = "ESP: ON"

    disconnectESPConnections()

    for _, targetPlayer in ipairs(Players:GetPlayers()) do
        addESPToPlayer(targetPlayer)

        table.insert(espConnections, targetPlayer.CharacterAdded:Connect(function()
            task.wait(0.5)
            addESPToPlayer(targetPlayer)
        end))

        table.insert(espConnections, targetPlayer:GetPropertyChangedSignal("Team"):Connect(function()
            task.defer(function()
                if espEnabled then
                    stopESP()
                    startESP()
                end
            end)
        end))
    end

    table.insert(espConnections, Players.PlayerAdded:Connect(function(targetPlayer)
        table.insert(espConnections, targetPlayer.CharacterAdded:Connect(function()
            task.wait(0.5)
            addESPToPlayer(targetPlayer)
        end))

        table.insert(espConnections, targetPlayer:GetPropertyChangedSignal("Team"):Connect(function()
            task.defer(function()
                if espEnabled then
                    stopESP()
                    startESP()
                end
            end)
        end))
    end))

    table.insert(espConnections, Players.PlayerRemoving:Connect(function(targetPlayer)
        removeESPFromPlayer(targetPlayer)
    end))

    table.insert(espConnections, RunService.Heartbeat:Connect(function()
        for _, targetPlayer in ipairs(Players:GetPlayers()) do
            addESPToPlayer(targetPlayer)
        end
    end))
end

local function refreshESP()
    if not espEnabled then return end

    stopESP()
    startESP()
end

    local blue = tonumber(espBlueBox.Text)

        espColor = Color3.fromRGB(
            math.clamp(math.floor(red), 0, 255),
            math.clamp(math.floor(green), 0, 255),
            math.clamp(math.floor(blue), 0, 255)
        )
        espRedBox.Text = tostring(math.floor(espColor.R * 255 + 0.5))
        espGreenBox.Text = tostring(math.floor(espColor.G * 255 + 0.5))
        espBlueBox.Text = tostring(math.floor(espColor.B * 255 + 0.5))
        refreshESP()

        espRedBox.Text = tostring(math.floor(espColor.R * 255 + 0.5))
        espGreenBox.Text = tostring(math.floor(espColor.G * 255 + 0.5))
        espBlueBox.Text = tostring(math.floor(espColor.B * 255 + 0.5))
    end
end

-- KEYBINDS
-- Press the configured keys to toggle speed, fly, and ESP.
------------------------------------------------
UIS.InputBegan:Connect(function(i, gpe)
    if gpe then return end
    if i.UserInputType ~= Enum.UserInputType.Keyboard then return end

    if bindingAction then
        if i.KeyCode == Enum.KeyCode.Backspace
            or i.KeyCode == Enum.KeyCode.Delete
            or i.KeyCode == Enum.KeyCode.Escape then
            keybinds[bindingAction] = nil
        elseif i.KeyCode ~= Enum.KeyCode.Unknown then
            keybinds[bindingAction] = i.KeyCode
        end

        bindingAction = nil
        updateKeybindButtons()
        return
    end

    if keybinds.Speed and i.KeyCode == keybinds.Speed then
        toggleSpeed()
    elseif keybinds.Fly and i.KeyCode == keybinds.Fly then
        toggleFly()
    elseif keybinds.ESP and i.KeyCode == keybinds.ESP then
        toggleESP()
    end
end)
